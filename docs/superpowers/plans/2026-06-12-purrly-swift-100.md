```swift
// ios-native/RaceGuide/Services/UsageLimits.swift
import Foundation

@MainActor
final class UsageLimits: ObservableObject {
    @Published var isPremium: Bool {
        didSet { defaults.set(isPremium, forKey: "usage_premium") }
    }
    @Published private(set) var aiMessagesThisMonth: Int

    private let defaults: UserDefaults
    private let monthKey = "usage_ai_month"
    private let countKey = "usage_ai_count"
    private let premiumKey = "usage_premium"

    static let freeAILimit = 15
    static let freeCatLimit = 1

    init(defaults: UserDefaults = .standard) {
        self.defaults = defaults
        self.isPremium = defaults.bool(forKey: "usage_premium")
        let storedMonth = defaults.string(forKey: "usage_ai_month") ?? ""
        let currentMonth = Self.currentMonthKey()
        if storedMonth == currentMonth {
            self.aiMessagesThisMonth = defaults.integer(forKey: "usage_ai_count")
        } else {
            self.aiMessagesThisMonth = 0
            defaults.set(currentMonth, forKey: "usage_ai_month")
            defaults.set(0, forKey: "usage_ai_count")
        }
    }

    var canSendAIMessage: Bool {
        isPremium || aiMessagesThisMonth < Self.freeAILimit
    }

    var aiMessagesRemaining: Int {
        isPremium ? Int.max : max(0, Self.freeAILimit - aiMessagesThisMonth)
    }

    func canAddCat(currentCount: Int) -> Bool {
        isPremium || currentCount < Self.freeCatLimit
    }

    func recordAIMessage() {
        aiMessagesThisMonth += 1
        defaults.set(aiMessagesThisMonth, forKey: countKey)
    }

    private static func currentMonthKey() -> String {
        let c = Calendar.current.dateComponents([.year, .month], from: Date())
        return "\(c.year ?? 0)-\(c.month ?? 0)"
    }
}
```

```swift
// ios-native/RaceGuide/Services/EmergencyStore.swift
import Foundation

@MainActor
final class EmergencyStore: ObservableObject {
    @Published private(set) var alerts: [EmergencyAlert] = []
    private let storageKey = "emergency_alerts"
    private let defaults: UserDefaults

    init(defaults: UserDefaults = .standard) {
        self.defaults = defaults
        load()
    }

    func load() {
        guard let data = defaults.data(forKey: storageKey),
              let decoded = try? JSONDecoder().decode([EmergencyAlert].self, from: data)
        else { return }
        alerts = decoded.sorted { $0.createdAt > $1.createdAt }
    }

    func add(_ alert: EmergencyAlert) {
        alerts.insert(alert, at: 0)
        persist()
    }

    func resolve(id: String) {
        guard let i = alerts.firstIndex(where: { $0.id == id }) else { return }
        alerts[i].isResolved = true
        persist()
    }

    func delete(id: String) {
        alerts.removeAll { $0.id == id }
        persist()
    }

    private func persist() {
        if let data = try? JSONEncoder().encode(alerts) {
            defaults.set(data, forKey: storageKey)
        }
    }
}
```

```swift
// ios-native/RaceGuide/Services/ExpenseStore.swift
import Foundation

@MainActor
final class ExpenseStore: ObservableObject {
    @Published private(set) var expenses: [Expense] = []
    private let storageKey = "expenses_data"
    private let defaults: UserDefaults

    init(defaults: UserDefaults = .standard) {
        self.defaults = defaults
        load()
    }

    func load() {
        guard let data = defaults.data(forKey: storageKey),
              let decoded = try? JSONDecoder().decode([Expense].self, from: data)
        else { return }
        expenses = decoded.sorted { $0.date > $1.date }
    }

    func add(_ expense: Expense) {
        expenses.insert(expense, at: 0)
        persist()
    }

    func delete(id: String) {
        expenses.removeAll { $0.id == id }
        persist()
    }

    func total(for category: ExpenseCategory) -> Double {
        expenses.filter { $0.category == category }.reduce(0) { $0 + $1.amount }
    }

    var grandTotal: Double { expenses.reduce(0) { $0 + $1.amount } }

    private func persist() {
        if let data = try? JSONEncoder().encode(expenses) {
            defaults.set(data, forKey: storageKey)
        }
    }
}
```

```swift
import Foundation

@MainActor
final class GamificationStore: ObservableObject {
    @Published private(set) var achievements: [Achievement] = Self.defaults
    @Published private(set) var totalXP: Int = 0

    private let storageKey = "gamification_data"
    private let defaults: UserDefaults

    init(defaults: UserDefaults = .standard) {
        self.defaults = defaults
        load()
    }

    func load() {
        guard let data = defaults.data(forKey: storageKey),
              let saved = try? JSONDecoder().decode([Achievement].self, from: data)
        else { recalculate(); return }
        achievements = saved
        totalXP = achievements.filter(\.isUnlocked).reduce(0) { $0 + $1.xpValue }
    }

    func unlock(id: String) {
        guard let i = achievements.firstIndex(where: { $0.id == id }),
              !achievements[i].isUnlocked else { return }
        achievements[i].isUnlocked = true
        achievements[i].unlockedAt = Date()
        totalXP += achievements[i].xpValue
        persist()
    }

    func checkAndUnlock(forCatCount count: Int) {
        if count >= 1 { unlock(id: "first_cat") }
        if count >= 3 { unlock(id: "three_cats") }
    }

    func checkAndUnlockScan() { unlock(id: "first_scan") }

    var level: Int { max(1, totalXP / 100 + 1) }
    var xpToNextLevel: Int { level * 100 - totalXP }

    private func recalculate() {
        totalXP = achievements.filter(\.isUnlocked).reduce(0) { $0 + $1.xpValue }
    }

    private func persist() {
        if let data = try? JSONEncoder().encode(achievements) {
            defaults.set(data, forKey: storageKey)
        }
    }

    static let defaults: [Achievement] = [
        Achievement(id: "first_cat",  title: "Premier compagnon",  description: "Ajoute ton premier chat",  icon: "heart.fill",    category: .cats,       xpValue: 50,  isUnlocked: false, unlockedAt: nil),
```

```swift
// ios-native/RaceGuide/Services/DiscussionsStore.swift
import Foundation

@MainActor
final class DiscussionsStore: ObservableObject {
    @Published private(set) var threads: [DiscussionThread] = []
    private let storageKey = "discussions_data"

    init() { load() }

    func load() {
        guard let data = UserDefaults.standard.data(forKey: storageKey),
              let decoded = try? JSONDecoder().decode([DiscussionThread].self, from: data)
        else { loadSamples(); return }
        threads = decoded
    }

    func addThread(title: String, firstMessage: String, authorName: String) {
        let msg = DiscussionMessage(id: UUID().uuidString, authorName: authorName,
                                    content: firstMessage, createdAt: Date())
        let thread = DiscussionThread(id: UUID().uuidString, title: title,
                                      authorName: authorName, createdAt: Date(), messages: [msg])
        threads.insert(thread, at: 0)
        persist()
    }

    func addMessage(_ text: String, author: String, toThread id: String) {
        guard let i = threads.firstIndex(where: { $0.id == id }) else { return }
        let msg = DiscussionMessage(id: UUID().uuidString, authorName: author,
                                    content: text, createdAt: Date())
        threads[i].messages.append(msg)
        persist()
    }

    private func persist() {
        if let data = try? JSONEncoder().encode(threads) {
            UserDefaults.standard.set(data, forKey: storageKey)
        }
    }

    private func loadSamples() {
        threads = [
            DiscussionThread(id: "1", title: "Mon Maine Coon griffe tout — conseils ?",
                             authorName: "Léa", createdAt: Date().addingTimeInterval(-7200),
                             messages: [
                                DiscussionMessage(id: "m1", authorName: "Léa",
                                                  content: "Il griffe les canapés, les rideaux… je suis à bout.",
```
